cmake_minimum_required(VERSION 3.28)
project(output-scheduler-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
enable_testing()
add_executable(schedule-model-test model-test.cpp)
target_include_directories(schedule-model-test PRIVATE ../../frontend)
target_link_libraries(schedule-model-test PRIVATE Qt6::Core)
add_test(NAME schedule-model COMMAND schedule-model-test)
add_executable(schedule-dialog-test dialog-test.cpp ../../frontend/widgets/OutputScheduleDialog.cpp)
set_target_properties(schedule-dialog-test PROPERTIES OUTPUT_NAME schedule-editor-test)
target_include_directories(schedule-dialog-test PRIVATE ../../frontend)
target_link_libraries(schedule-dialog-test PRIVATE Qt6::Widgets)
target_compile_definitions(schedule-dialog-test PRIVATE SCHEDULER_FRONTEND="${CMAKE_CURRENT_LIST_DIR}/../../frontend")
add_test(NAME schedule-dialog COMMAND schedule-dialog-test)
set_tests_properties(schedule-dialog PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal" TIMEOUT 30)
