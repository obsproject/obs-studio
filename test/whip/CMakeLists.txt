cmake_minimum_required(VERSION 3.28)
project(whip_ice_regression LANGUAGES C)

if(NOT IS_DIRECTORY "${LIBJUICE_SOURCE_DIR}/src")
  message(FATAL_ERROR "Set LIBJUICE_SOURCE_DIR to the patched libjuice checkout")
endif()
set(NO_TESTS ON CACHE BOOL "Disable libjuice's network test suite" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build static test dependency" FORCE)
add_subdirectory("${LIBJUICE_SOURCE_DIR}" juice)
add_executable(whip-ice-test ice-roles.c "${LIBJUICE_SOURCE_DIR}/test/stun.c")
target_include_directories(whip-ice-test PRIVATE "${LIBJUICE_SOURCE_DIR}/src" "${LIBJUICE_SOURCE_DIR}/include/juice")
target_link_libraries(whip-ice-test PRIVATE juice-static)
if(WIN32)
  target_link_libraries(whip-ice-test PRIVATE ws2_32)
endif()
enable_testing()
add_test(NAME whip-ice-roles COMMAND whip-ice-test)
